import React, { useState } from 'react';
import { View, ScrollView, StyleSheet, Button } from 'react-native';
import { Text, Card, Title, Appbar, DataTable, Chip, Paragraph } from 'react-native-paper';
import { LineChart, Grid, YAxis, XAxis, BarChart, PieChart } from 'react-native-svg-charts';
import * as shape from 'd3-shape';
import RNFS from 'react-native-fs';
import Share from 'react-native-share';
import RNHTMLtoPDF from 'react-native-html-to-pdf';

export default function EmendasDashboard() {
  const [filtroMunicipio, setFiltroMunicipio] = useState(null);
  const [filtroTipo, setFiltroTipo] = useState(null);

  const anos = ['2019', '2020', '2021', '2022', '2023', '2024'];
  const dados = [42, 56, 68, 71, 89, 96];

  const municipios = ['Recife', 'Caruaru', 'Petrolina', 'Olinda'];
  const valores = [60, 40, 35, 30];

  const tipos = ['Saúde', 'Educação', 'Infraestrutura', 'Outros'];

  const pieData = [
    { key: 1, value: 45, svg: { fill: '#f44336' }, label: 'Saúde' },
    { key: 2, value: 25, svg: { fill: '#2196f3' }, label: 'Educação' },
    { key: 3, value: 20, svg: { fill: '#ff9800' }, label: 'Infraestrutura' },
    { key: 4, value: 10, svg: { fill: '#4caf50' }, label: 'Outros' },
  ];

  const emendas = [
    { numero: 1022, objeto: "Políticas de saúde", cidade: "Recife", status: "Em execução", cor: "#2e7d32", tipo: "Saúde" },
    { numero: 1021, objeto: "Infraestrutura urbana", cidade: "Caruaru", status: "Em execução", cor: "#2e7d32", tipo: "Infraestrutura" },
    { numero: 1014, objeto: "Equipamentos", cidade: "Petrolina", status: "Em análise", cor: "#ef6c00", tipo: "Educação" },
    { numero: 1008, objeto: "Equipamentos", cidade: "Olinda", status: "Em execução", cor: "#2e7d32", tipo: "Educação" },
  ];

  const emendasFiltradas = emendas.filter(e =>
    (!filtroMunicipio || e.cidade === filtroMunicipio) &&
    (!filtroTipo || e.tipo === filtroTipo)
  );

  const exportCSV = async () => {
    const headers = "Número,Objeto,Cidade,Status\n";
    const rows = emendasFiltradas.map(e => [e.numero, e.objeto, e.cidade, e.status]);
    const csvContent = headers + rows.map(e => e.join(",")).join("\n");

    const path = `${RNFS.DocumentDirectoryPath}/emendas.csv`;
    await RNFS.writeFile(path, csvContent, 'utf8');
    await Share.open({ url: `file://${path}`, type: 'text/csv' });
  };

  const exportPDF = async () => {
    const htmlRows = emendasFiltradas.map(e => `<tr><td>${e.numero}</td><td>${e.objeto}</td><td>${e.cidade}</td><td>${e.status}</td></tr>`).join('');
    const htmlContent = `
      <h1>Relatório de Emendas</h1>
      <table border="1" style="width: 100%">
        <tr><th>Número</th><th>Objeto</th><th>Cidade</th><th>Status</th></tr>
        ${htmlRows}
      </table>
    `;

    const options = {
      html: htmlContent,
      fileName: 'relatorio_emendas',
      directory: 'Documents',
    };

    const file = await RNHTMLtoPDF.convert(options);
    await Share.open({ url: `file://${file.filePath}`, type: 'application/pdf' });
  };

  return (
    <>
      <Appbar.Header>
        <Appbar.Content title="Gestão de Emendas" subtitle="Senador Humberto Costa" />
      </Appbar.Header>

      <ScrollView style={styles.container}>
        <View style={styles.statsContainer}>
          <Card style={styles.card}><Card.Content><Title>Emendas</Title><Text variant="headlineMedium">256</Text></Card.Content></Card>
          <Card style={styles.card}><Card.Content><Title>Em Execução</Title><Text style={{ color: '#2e7d32' }}>72</Text></Card.Content></Card>
          <Card style={styles.card}><Card.Content><Title>Em Análise</Title><Text style={{ color: '#ef6c00' }}>45</Text></Card.Content></Card>
        </View>

        <View style={styles.filterContainer}>
          {municipios.map((m, i) => (
            <Chip key={i} selected={filtroMunicipio === m} onPress={() => setFiltroMunicipio(m)} style={{ margin: 4 }}>{m}</Chip>
          ))}
          {tipos.map((t, i) => (
            <Chip key={t} selected={filtroTipo === t} onPress={() => setFiltroTipo(t)} style={{ margin: 4 }}>{t}</Chip>
          ))}
          <Chip onPress={() => { setFiltroMunicipio(null); setFiltroTipo(null); }} style={{ margin: 4 }} icon="close">Limpar</Chip>
        </View>

        <Card style={styles.card}><Card.Content><Title>Emendas por Ano</Title>
          <View style={{ height: 200, flexDirection: 'row', paddingVertical: 10 }}>
            <YAxis data={dados} contentInset={{ top: 20, bottom: 20 }} svg={{ fontSize: 10, fill: 'grey' }} />
            <View style={{ flex: 1, marginLeft: 10 }}>
              <LineChart
                style={{ flex: 1 }}
                data={dados}
                svg={{ stroke: '#3f51b5' }}
                contentInset={{ top: 20, bottom: 20 }}
                curve={shape.curveMonotoneX}
              >
                <Grid />
              </LineChart>
              <XAxis
                style={{ marginHorizontal: -10, height: 20 }}
                data={dados}
                formatLabel={(value, index) => anos[index]}
                contentInset={{ left: 15, right: 15 }}
                svg={{ fontSize: 10, fill: 'grey' }}
              />
            </View>
          </View>
        </Card.Content></Card>

        <Card style={styles.card}><Card.Content><Title>Distribuição por Município</Title>
          <BarChart style={{ height: 200 }} data={valores} svg={{ fill: '#4caf50' }} contentInset={{ top: 30, bottom: 30 }} spacingInner={0.2} gridMin={0} />
          <XAxis style={{ marginTop: 10 }} data={valores} formatLabel={(value, index) => municipios[index]} contentInset={{ left: 10, right: 10 }} svg={{ fontSize: 10, fill: 'black' }} />
        </Card.Content></Card>

        <Card style={styles.card}><Card.Content><Title>Distribuição por Tipo</Title>
          <PieChart style={{ height: 200 }} data={pieData} />
        </Card.Content></Card>

        <Card style={styles.card}><Card.Content><Title>Exportar Dados</Title>
          <View style={{ flexDirection: 'row', justifyContent: 'space-around', marginTop: 10 }}>
            <Button title="Exportar CSV" onPress={exportCSV} />
            <Button title="Exportar PDF" onPress={exportPDF} />
          </View>
        </Card.Content></Card>

        <Card style={styles.card}><Card.Content><Title>Últimas Emendas</Title>
          <DataTable>
            <DataTable.Header>
              <DataTable.Title>Número</DataTable.Title>
              <DataTable.Title>Objeto</DataTable.Title>
              <DataTable.Title>Município</DataTable.Title>
              <DataTable.Title>Status</DataTable.Title>
            </DataTable.Header>

            {emendasFiltradas.map((emenda, index) => (
              <DataTable.Row key={index}>
                <DataTable.Cell>{emenda.numero}</DataTable.Cell>
                <DataTable.Cell>{emenda.objeto}</DataTable.Cell>
                <DataTable.Cell>{emenda.cidade}</DataTable.Cell>
                <DataTable.Cell><Chip style={{ backgroundColor: emenda.cor }} textStyle={{ color: '#fff' }}>{emenda.status}</Chip></DataTable.Cell>
              </DataTable.Row>
            ))}
          </DataTable>
        </Card.Content></Card>
      </ScrollView>
    </>
  );
}

const styles = StyleSheet.create({
  container: { padding: 10, backgroundColor: '#f4f4f4' },
  statsContainer: { flexDirection: 'row', justifyContent: 'space-between', marginBottom: 10 },
  card: { marginVertical: 10, backgroundColor: 'white' },
  filterContainer: { flexDirection: 'row', flexWrap: 'wrap', marginVertical: 10, paddingHorizontal: 5 },
});
